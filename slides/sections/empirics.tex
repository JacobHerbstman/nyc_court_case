\begin{frame}[t]
    \frametitle{Data}
    \small

    \begin{itemize}
        \item \textbf{Unit}: 2010 City Council districts
        \begin{itemize}
            \footnotesize
            \item 1990 council districts do not exist online, will need to be digitized from paper maps
        \end{itemize}

        \item \textbf{Treatment}: 1990 homeownership from NHGIS tenure counts
        \begin{itemize}
            \footnotesize
            \item Area-weight 1990 tracts into 2010 Council districts.
            \item Standardize within assigned majority borough.
        \end{itemize}

        \item \textbf{Outcomes}: Primary Land Use Tax Lot Output (PLUTO) dataset
        \begin{itemize}
            \footnotesize
            \item New housing units constructed from 1980-2025
            \item Allows me to observe the zoning, ownership, and assessed value of every lot in the city
            \item Main flaw: if a building is demolished and rebuilt, I will not be able to see the old building in the data
        \end{itemize}
        \item Plan to use the Zoning Application Portal (ZAP) dataset, Uniform Land-Use Review Procedure (ULURP) dataset, and city council ordinance records
        to quantify member deference directly in the future
    \end{itemize}
\end{frame}

\begin{frame}[t]
    \frametitle{Measuring Homeowner Exposure}
    \small

    \only<1>{
        \begin{itemize}
            \item Homeownership rates are a good proxy for local resistance to new housing supply \citep{fischel_homevoter_2009,ortalo-magne_political_2014,khan_decentralized_2021}
            \item Let \(h_j\) be district \(j\)'s raw 1990 homeownership share.
            \item Define homeowner exposure as:
        \end{itemize}

        \vspace{0.6em}
        \[
            H_j
            =
            \frac{
                h_j
                -
                \bar h_{b(j)}
            }{
                s_{b(j)}
            }
        \]

        \vspace{0.2em}
        \begin{itemize}
            \footnotesize
            \item \(b(j)\): district \(j\)'s assigned majority borough.
            \item \(\bar h_{b(j)}\) and \(s_{b(j)}\): mean and standard deviation of \(h_j\) among Council districts in the same borough.
        \end{itemize}
    }

    \only<2>{
        \vspace{-1.0em}
        \begin{center}
            \includegraphics[page=1,height=0.82\textheight,trim={0.25in 0in 0.25in 0in},clip]{../tasks/build_ccd2010_homeownership_1990_measure/output/ccdist2010_homeownership_1990_map.pdf}
        \end{center}
    }
\end{frame}
