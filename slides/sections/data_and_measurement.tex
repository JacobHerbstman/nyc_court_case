\begin{frame}
    \frametitle{Measuring Local Opposition}

    \textbf{Homeowner exposure \(H_j\)}: 1990 homeownership share in 2010 City Council districts
    \begin{itemize}
        \item Project 1990 tract tenure counts into fixed 2010 Council districts
        \item Compare each district's homeownership share to its assigned-borough average
        \item Standardize treatment within assigned majority borough
    \end{itemize}

    \vspace{0.5em}
    \textbf{Interpretation}
    \begin{itemize}
        \item High-\(H_j\) districts are homeowner-heavy relative to politically comparable districts in the same assigned borough
        \item Within-assigned-borough standardization avoids comparing Staten Island and Manhattan as if they faced the same development baseline
    \end{itemize}
\end{frame}
